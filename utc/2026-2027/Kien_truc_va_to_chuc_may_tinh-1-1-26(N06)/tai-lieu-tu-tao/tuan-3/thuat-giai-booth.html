// Booth's Algorithm – Signed Binary Multiplication (C++17)
#include <iostream>
#include <bitset>
#include <cstdint>

using namespace std;

// Mở rộng dấu cho giá trị x với độ dài n bit
int64_t signExtend(int64_t x, int n) {
    int64_t mask = (1LL << n) - 1;
    if (x & (1LL << (n - 1))) {
        x |= ~mask;   // set các bit cao hơn thành 1
    } else {
        x &= mask;    // giữ nguyên n bit thấp
    }
    return x;
}

// Thuật toán Booth
int64_t boothMultiply(int n, int64_t multiplicand, int64_t multiplier) {
    int64_t mask = (1LL << n) - 1;

    // Khởi tạo các thanh ghi
    int64_t M = signExtend(multiplicand & mask, n);   // M = Multiplicand
    int64_t Q = signExtend(multiplier & mask, n);     // Q = Multiplier
    int64_t A = 0;                                   // A = 0
    int Q0_ext = 0;                                  // Q_{-1} = 0
    int count = n;

    while (count > 0) {
        int q0 = Q & 1;      // bit thấp nhất của Q

        // Kiểm tra cặp bit (q0, Q0_ext)
        if (q0 == 0 && Q0_ext == 1) {
            // 01 : A = A + M
            A = A + M;
            A = signExtend(A & mask, n);
        } else if (q0 == 1 && Q0_ext == 0) {
            // 10 : A = A - M
            A = A - M;
            A = signExtend(A & mask, n);
        }
        // Còn lại 00 hoặc 11: không làm gì

        // Dịch phải số học (A, Q, Q0_ext)
        int sign = (A >> (n - 1)) & 1;   // bit dấu của A trước khi dịch
        int low_A = A & 1;               // bit thấp nhất của A
        int old_Q0 = Q & 1;              // bit thấp nhất của Q (cũ)

        // Dịch phải A: giữ nguyên bit dấu
        A = (A >> 1) | (sign << (n - 1));
        A = signExtend(A & mask, n);

        // Dịch phải Q, đưa bit thấp nhất của A cũ vào bit cao nhất của Q
        Q = (Q >> 1) | (low_A << (n - 1));
        Q &= mask;   // giữ n bit

        // Cập nhật Q0_ext = old_Q0
        Q0_ext = old_Q0;

        count--;
    }

    // Kết quả là (A, Q) tạo thành số nhị phân 2n bit
    int64_t result = (A << n) | (Q & mask);

    // Mở rộng dấu cho kết quả 2n bit
    if (result & (1LL << (2 * n - 1))) {
        result |= ~((1LL << (2 * n)) - 1);
    }

    return result;
}

int main() {
    int n;
    int64_t multiplicand, multiplier;

    cout << "Nhap so bit n: ";
    cin >> n;
    cout << "Nhap so bi nhan (Multiplicand): ";
    cin >> multiplicand;
    cout << "Nhap so nhan (Multiplier): ";
    cin >> multiplier;

    int64_t product = boothMultiply(n, multiplicand, multiplier);

    cout << "Tich = " << product << endl;

    // In biểu diễn nhị phân (2n bit)
    cout << "Nhi phan (2n bit): ";
    int totalBits = 2 * n;
    if (totalBits <= 64) {
        for (int i = totalBits - 1; i >= 0; --i) {
            cout << ((product >> i) & 1);
            if (i % 4 == 0 && i != 0) cout << " ";
        }
        cout << endl;
    } else {
        cout << "Khong the in do qua 64 bit." << endl;
    }

    return 0;
}
